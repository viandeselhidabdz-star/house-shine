import React, { useState } from "react";

// Maison Éclat — Single-file React component (Tailwind CSS)
// How to use:
// 1. Create a React app (Vite/Create React App) and install Tailwind CSS.
// 2. Drop this component in src/App.jsx and ensure Tailwind is configured.
// 3. This is a static demo (mock products). Replace mock data with your API.

export default function MaisonEclatSite() {
  const SITE_NAME = "Maison Éclat"; // Chic name

  const initialProducts = [
    {
      id: 1,
      title: "Sac " + SITE_NAME + " Mini",
      price: 129.0,
      desc: "Cuir végétal, finitions dorées, format compact.",
      image:
        "https://images.unsplash.com/photo-1520975916427-5b8b3b2f7d9b?auto=format&fit=crop&w=800&q=60",
    },
    {
      id: 2,
      title: "Montre Élégante",
      price: 249.0,
      desc: "Mécanisme quartz précis, bracelet en cuir premium.",
      image:
        "https://images.unsplash.com/photo-1519744792095-2f2205e87b6f?auto=format&fit=crop&w=800&q=60",
    },
    {
      id: 3,
      title: "Écharpe Soie Luxe",
      price: 79.0,
      desc: "Soie naturelle imprimée, motif exclusif.",
      image:
        "https://images.unsplash.com/photo-1520975916427-5b8b3b2f7d9b?auto=format&fit=crop&w=800&q=60",
    },
  ];

  const [products] = useState(initialProducts);
  const [cart, setCart] = useState([]);
  const [query, setQuery] = useState("");
  const [showCheckout, setShowCheckout] = useState(false);
  const [orderComplete, setOrderComplete] = useState(false);
  const [customer, setCustomer] = useState({ name: "", email: "", address: "" });

  function addToCart(product) {
    setCart((c) => {
      const found = c.find((i) => i.id === product.id);
      if (found) return c.map((i) => (i.id === product.id ? { ...i, qty: i.qty + 1 } : i));
      return [...c, { ...product, qty: 1 }];
    });
  }

  function removeFromCart(id) {
    setCart((c) => c.filter((i) => i.id !== id));
  }

  function changeQty(id, qty) {
    if (qty <= 0) return removeFromCart(id);
    setCart((c) => c.map((i) => (i.id === id ? { ...i, qty } : i)));
  }

  function total() {
    return cart.reduce((s, p) => s + p.price * p.qty, 0).toFixed(2);
  }

  function handleCheckout(e) {
    e.preventDefault();
    // Mock checkout success
    setOrderComplete(true);
    setCart([]);
    setShowCheckout(false);
  }

  const filtered = products.filter(
    (p) =>
      p.title.toLowerCase().includes(query.toLowerCase()) ||
      p.desc.toLowerCase().includes(query.toLowerCase())
  );

  return (
    <div className="min-h-screen bg-gray-50 text-gray-800">
      {/* NAVBAR */}
      <header className="bg-white shadow">
        <div className="container mx-auto px-6 py-4 flex items-center justify-between">
          <div className="flex items-center space-x-3">
            <div className="rounded-full bg-gray-900 text-white w-12 h-12 flex items-center justify-center text-xl font-bold">
              M
            </div>
            <div>
              <div className="text-lg font-semibold">{SITE_NAME}</div>
              <div className="text-xs text-gray-500">Luxe. Simplicité. Éclat.</div>
            </div>
          </div>

          <div className="flex-1 mx-6">
            <input
              aria-label="Recherche"
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder="Rechercher un produit, ex: montre"
              className="w-full rounded-lg border px-3 py-2"
            />
          </div>

          <div className="flex items-center space-x-4">
            <button
              onClick={() => setShowCheckout((s) => !s)}
              className="px-4 py-2 rounded-md border hover:bg-gray-100"
            >
              Panier ({cart.reduce((s, i) => s + i.qty, 0)})
            </button>
            <a href="#contact" className="text-sm text-gray-600 hover:underline">
              Contact
            </a>
          </div>
        </div>
      </header>

      {/* HERO */}
      <section className="container mx-auto px-6 py-12 flex flex-col md:flex-row items-center gap-8">
        <div className="flex-1">
          <h1 className="text-4xl md:text-5xl font-extrabold">Bienvenue chez {SITE_NAME}</h1>
          <p className="mt-4 text-gray-600 max-w-xl">
            Une sélection raffinée d'accessoires et pièces uniques. Livraison sous 3-7 jours ouvrés.
          </p>

          <div className="mt-6 flex items-center gap-3">
            <button
              onClick={() => window.scrollTo({ top: 1000, behavior: "smooth" })}
              className="px-6 py-3 rounded-lg bg-gray-900 text-white font-medium shadow"
            >
              Voir la boutique
            </button>
            <a href="#contact" className="text-sm text-gray-600 hover:underline">
              Besoin d'aide ?
            </a>
          </div>
        </div>

        <div className="flex-1">
          <div className="rounded-xl overflow-hidden shadow-lg">
            <img
              src="https://images.unsplash.com/photo-1503342217505-b0a15ec3261c?auto=format&fit=crop&w=1200&q=60"
              alt="Hero"
              className="w-full h-80 object-cover"
            />
          </div>
        </div>
      </section>

      {/* PRODUCTS */}
      <section className="container mx-auto px-6 py-8">
        <h2 className="text-2xl font-bold mb-4">Notre sélection</h2>
        <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-6">
          {filtered.map((p) => (
            <article key={p.id} className="bg-white rounded-lg shadow p-4 flex flex-col">
              <div className="h-48 w-full overflow-hidden rounded-md">
                <img src={p.image} alt={p.title} className="w-full h-full object-cover" />
              </div>
              <div className="mt-3 flex-1">
                <h3 className="font-semibold">{p.title}</h3>
                <p className="text-sm text-gray-500 mt-1">{p.desc}</p>
              </div>
              <div className="mt-3 flex items-center justify-between">
                <div className="text-lg font-bold">€{p.price.toFixed(2)}</div>
                <div className="flex items-center gap-2">
                  <button
                    onClick={() => addToCart(p)}
                    className="px-3 py-2 rounded-md border hover:bg-gray-50"
                  >
                    Ajouter
                  </button>
                </div>
              </div>
            </article>
          ))}
        </div>
      </section>

      {/* SIDEBAR CHECKOUT */}
      <aside
        className={`fixed right-4 top-20 w-80 bg-white rounded-xl shadow-lg p-4 transition-transform transform ${
          showCheckout ? "translate-y-0" : "translate-y-[-10%] scale-95 opacity-0 pointer-events-none"
        }`}
        aria-hidden={!showCheckout}
      >
        <h4 className="font-semibold">Votre panier</h4>
        <div className="mt-3 space-y-3 max-h-64 overflow-auto">
          {cart.length === 0 && <div className="text-sm text-gray-500">Panier vide</div>}
          {cart.map((item) => (
            <div key={item.id} className="flex items-center gap-3">
              <img src={item.image} alt={item.title} className="w-12 h-12 object-cover rounded" />
              <div className="flex-1">
                <div className="text-sm font-medium">{item.title}</div>
                <div className="text-xs text-gray-500">€{item.price.toFixed(2)}</div>
                <div className="mt-1 flex items-center gap-2">
                  <button onClick={() => changeQty(item.id, item.qty - 1)} className="px-2 border rounded">-</button>
                  <div className="text-sm">{item.qty}</div>
                  <button onClick={() => changeQty(item.id, item.qty + 1)} className="px-2 border rounded">+</button>
                </div>
              </div>
              <button onClick={() => removeFromCart(item.id)} className="text-xs text-red-500">Suppr</button>
            </div>
          ))}
        </div>

        <div className="mt-4 border-t pt-3">
          <div className="flex items-center justify-between">
            <div className="text-sm text-gray-600">Total</div>
            <div className="font-semibold">€{total()}</div>
          </div>

          <div className="mt-3">
            <button
              onClick={() => setShowCheckout(false)}
              className="w-full px-4 py-2 rounded-md border mb-2"
            >
              Continuer mes achats
            </button>
            <button
              onClick={() => {
                setShowCheckout(true);
                // Smooth open the checkout modal
                const el = document.getElementById("checkout-modal");
                if (el) el.scrollIntoView({ behavior: "smooth" });
              }}
              disabled={cart.length === 0}
              className={w-full px-4 py-2 rounded-md ${cart.length === 0 ? "bg-gray-200 text-gray-500" : "bg-gray-900 text-white"}}>
              Passer à la caisse
            </button>
          </div>
        </div>
      </aside>

      {/* CHECKOUT SECTION (simple) */}
      <section id="checkout-modal" className="container mx-auto px-6 py-12">
        <div className="max-w-3xl mx-auto bg-white rounded-xl shadow p-6">
          <h3 className="text-xl font-semibold">Paiement & Livraison</h3>
          {orderComplete && (
            <div className="mt-4 p-4 bg-green-50 border border-green-100 rounded">Merci ! Votre commande a été reçue.</div>
          )}

          <form onSubmit={handleCheckout} className="mt-4 grid grid-cols-1 md:grid-cols-2 gap-4">
            <input
              required
              placeholder="Nom complet"
              value={customer.name}
              onChange={(e) => setCustomer((c) => ({ ...c, name: e.target.value }))}
              className="border rounded px-3 py-2"
            />
            <input
              required
              placeholder="Email"
              type="email"
              value={customer.email}
              onChange={(e) => setCustomer((c) => ({ ...c, email: e.target.value }))}
              className="border rounded px-3 py-2"
            />
            <input
              required
              placeholder="Adresse de livraison"
              value={customer.address}
              onChange={(e) => setCustomer((c) => ({ ...c, address: e.target.value }))}
              className="border rounded px-3 py-2 md:col-span-2"
            />

            <div className="md:col-span-2 flex items-center justify-between">
              <div>
                <div className="text-sm text-gray-600">Total à payer</div>
                <div className="text-lg font-semibold">€{total()}</div>
              </div>
              <button
                type="submit"
                disabled={cart.length === 0}
                className={px-6 py-2 rounded-md ${cart.length === 0 ? "bg-gray-200 text-gray-500" : "bg-gray-900 text-white"}}>
                Confirmer la commande
              </button>
            </div>
          </form>
        </div>
      </section>

      {/* CONTACT & FOOTER */}
      <footer id="contact" className="bg-white border-t mt-8">
        <div className="container mx-auto px-6 py-10 grid grid-cols-1 md:grid-cols-3 gap-6">
          <div>
            <h4 className="font-semibold text-lg">{SITE_NAME}</h4>
            <p className="text-sm text-gray-600 mt-2">Adresse fictive • Service client: +213 6X XX XX XX</p>
          </div>

          <div>
            <h5 className="font-medium">Liens</h5>
            <ul className="mt-3 text-sm text-gray-600 space-y-2">
              <li>Boutique</li>
              <li>À propos</li>
              <li>Livraison & retours</li>
            </ul>
          </div>

          <div>
            <h5 className="font-medium">Contactez-nous</h5>
            <form className="mt-3 space-y-2" onSubmit={(e) => e.preventDefault()}>
              <input placeholder="Nom" className="w-full border rounded px-3 py-2" />
              <input placeholder="Email" className="w-full border rounded px-3 py-2" />
              <textarea placeholder="Message" className="w-full border rounded px-3 py-2" rows={3} />
              <button className="px-4 py-2 rounded bg-gray-900 text-white">Envoyer</button>
            </form>
          </div>
        </div>

        <div className="text-center text-xs text-gray-500 py-4">© {new Date().getFullYear()} {SITE_NAME} — Tous droits réservés</div>
      </footer>
    </div>
  );
}
